""" Text → Image → Video Generator ============================== One complete script that turns a text prompt into an image and then into a short video. Models used: - Text-to-Image : Stable Diffusion XL - Image-to-Video: Stable Video Diffusion XT -------------------------------------------------- INSTALL (run once): pip install torch torchvision diffusers transformers accelerate pillow Optional (better speed/VRAM): pip install xformers -------------------------------------------------- USAGE EXAMPLES: python text_to_image_to_video.py --prompt "a cat wearing sunglasses on a beach" python text_to_image_to_video.py \ --prompt "cyberpunk city at night, neon lights, rain" \ --seed 42 \ --motion 150 \ --num_frames 25 python text_to_image_to_video.py --prompt "a red sports car on a mountain road" --device cpu """ import argparse from pathlib import Path import torch from diffusers import AutoPipelineForText2Image, StableVideoDiffusionPipeline from diffusers.utils import export_to_video def parse_args(): parser = argparse.ArgumentParser( description="Generate an image from text and turn it into a video" ) parser.add_argument( "--prompt", type=str, required=True, help="Text prompt for the image", ) parser.add_argument( "--negative_prompt", type=str, default="blurry, low quality, deformed, ugly, text, watermark, bad anatomy", help="Negative prompt", ) parser.add_argument("--seed", type=int, default=42, help="Random seed") parser.add_argument("--steps", type=int, default=28, help="Image inference steps") parser.add_argument("--guidance", type=float, default=7.0, help="Guidance scale") parser.add_argument( "--motion", type=int, default=127, help="Motion intensity (1-255). Higher = more movement", ) parser.add_argument( "--num_frames", type=int, default=25, help="Number of video frames" ) parser.add_argument("--fps", type=int, default=7, help="Output video FPS") parser.add_argument( "--output_dir", type=str, default="outputs", help="Folder to save results" ) parser.add_argument( "--device", type=str, default="cuda", choices=["cuda", "mps", "cpu"], help="Device to run on", ) return parser.parse_args() def main(): args = parse_args() # Create output folder output_dir = Path(args.output_dir) output_dir.mkdir(parents=True, exist_ok=True) # Device & dtype device = args.device if device == "cuda" and not torch.cuda.is_available(): print("CUDA not available → falling back to CPU (very slow)") device = "cpu" dtype = torch.float16 if device != "cpu" else torch.float32 generator = torch.Generator(device=device).manual_seed(args.seed) print("=" * 60) print("1/3 Loading Text-to-Image model (SDXL)...") print("=" * 60) t2i = AutoPipelineForText2Image.from_pretrained( "stabilityai/stable-diffusion-xl-base-1.0", torch_dtype=dtype, variant="fp16" if device != "cpu" else None, use_safetensors=True, ) t2i.enable_model_cpu_offload() print("\n2/3 Generating image from prompt...") print(f" Prompt: {args.prompt}") image = t2i( prompt=args.prompt, negative_prompt=args.negative_prompt, num_inference_steps=args.steps, guidance_scale=args.guidance, height=1024, width=1024, generator=generator, ).images[0] image_path = output_dir / "generated_image.png" image.save(image_path) print(f" Image saved → {image_path}") # Resize for Stable Video Diffusion (recommended resolution) image = image.resize((1024, 576)) print("\n3/3 Loading Image-to-Video model (SVD-XT)...") print("=" * 60) i2v = StableVideoDiffusionPipeline.from_pretrained( "stabilityai/stable-video-diffusion-img2vid-xt", torch_dtype=dtype, variant="fp16" if device != "cpu" else None, ) i2v.enable_model_cpu_offload() print(" Generating video... (this may take a minute)") frames = i2v( image, decode_chunk_size=8, # lower = less VRAM generator=generator, num_frames=args.num_frames, motion_bucket_id=args.motion, noise_aug_strength=0.02, ).frames[0] video_path = output_dir / "generated_video.mp4" export_to_video(frames, str(video_path), fps=args.fps) print("\n" + "=" * 60) print("DONE!") print(f"Image → {image_path}") print(f"Video → {video_path}") print("=" * 60) if __name__ == "__main__": main()
