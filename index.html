""" Text → Image → Video (Fixed Version) ==================================== """ import argparse from pathlib import Path import torch from diffusers import AutoPipelineForText2Image, StableVideoDiffusionPipeline from diffusers.utils import export_to_video from huggingface_hub import login def parse_args(): parser = argparse.ArgumentParser() parser.add_argument("--prompt", type=str, required=True) parser.add_argument("--negative_prompt", type=str, default="blurry, low quality, deformed, ugly, text, watermark") parser.add_argument("--seed", type=int, default=42) parser.add_argument("--steps", type=int, default=25) parser.add_argument("--guidance", type=float, default=7.0) parser.add_argument("--motion", type=int, default=127) parser.add_argument("--num_frames", type=int, default=14) # safer default (was 25) parser.add_argument("--fps", type=int, default=7) parser.add_argument("--output_dir", type=str, default="outputs") parser.add_argument("--token", type=str, default=None, help="Hugging Face token") return parser.parse_args() def main(): args = parse_args() output_dir = Path(args.output_dir) output_dir.mkdir(parents=True, exist_ok=True) # Login if token provided if args.token: login(token=args.token) else: print("Tip: If you get permission errors, run with --token YOUR_HF_TOKEN") print("Or run: huggingface-cli login") device = "cuda" if torch.cuda.is_available() else "cpu" dtype = torch.float16 if device == "cuda" else torch.float32 generator = torch.Generator(device=device).manual_seed(args.seed) print("\n=== 1. Loading Text-to-Image (SDXL) ===") try: t2i = AutoPipelineForText2Image.from_pretrained( "stabilityai/stable-diffusion-xl-base-1.0", torch_dtype=dtype, variant="fp16" if device == "cuda" else None, use_safetensors=True, ) t2i.enable_model_cpu_offload() except Exception as e: print("Error loading SDXL:", e) print("Falling back to smaller model...") t2i = AutoPipelineForText2Image.from_pretrained( "stabilityai/sd-turbo", torch_dtype=dtype, variant="fp16" if device == "cuda" else None, ) t2i.enable_model_cpu_offload() print("Generating image...") image = t2i( prompt=args.prompt, negative_prompt=args.negative_prompt, num_inference_steps=args.steps, guidance_scale=args.guidance, height=1024, width=1024, generator=generator, ).images[0] image_path = output_dir / "generated_image.png" image
